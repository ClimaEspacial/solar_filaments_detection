\begin{frame}
\frametitle{Algorithm for Storm Identification (using SYM-H)}

\begin{itemize}
    \item \textbf{Storm Identification Trigger:} An event is classified as a storm when the SYM-H index reaches \textbf{$-50~\textrm{nT}$} or less.

   
    \item \textbf{Main Phase Start:} Begins when the SYM-H index reaches the quiet levels prior storm. 

      \item \textbf{Initial Phase Start:} Defined as the point where the index reaches quiet levels \textit{before} its local maximum (the peak before the main phase drop).

    
    \item \textbf{Recovery Phase Start:} This phase begins exactly at the \textbf{minimum SYM-H value} (the storm's peak).

    \item \textbf{Recovery Phase End:} The storm is considered over when the index returns to \textbf{20\%} of its minimum value OR reaches a quiet level of \textbf{$-15~\textrm{nT}$}.
\end{itemize}

\end{frame}

        \item High-Speed Streams (HSSs)
    \end{itemize}

    \begin{itemize}
        \item Public catalogs:
        \begin{itemize}
            \item Cane and Richardson Lists

        \item Visual analysis of solar wind data (NASA/GSFC OMNIWeb)
    \end{itemize}

        \item ICME-driven: 308
        \item HSS-driven: 315
    \end{itemize}
